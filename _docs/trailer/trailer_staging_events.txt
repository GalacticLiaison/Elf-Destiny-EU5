# Instant Elven conversion (mirrors elf_destiny_story.22 "convert" + .23 rewards)
elf_trailer.1 = {
	title = no_title
	desc = no_desc
	outcome = neutral
	type = country_event
	hidden = yes

	immediate = {
		change_culture = culture:elven_culture
		every_character = {
			change_character_culture = culture:elven_culture
		}
		every_pop = {
			change_pop_culture = culture:elven_culture
		}
		every_province = {
			every_location_in_province = {
				change_institution_progress = {
					type = institution:elven
					value = 100
				}
			}
		}

		set_parliament_type = parliament_type:elven_parliament

		add_country_modifier = {
			modifier = realm_lord
			mode = add_and_extend
			desc = you_are_worthy
		}

		set_global_variable = {
			name = expeditions_unlocked
			value = yes
		}
	}
}

# Grand Portal restored in capital (mirrors elf_destiny_story.51/.52/.53)
elf_trailer.3 = {
	title = no_title
	desc = no_desc
	outcome = neutral
	type = country_event
	hidden = yes

	immediate = {
		capital = {
			set_variable = grand_portal_eligible
			destroy_all_buildings_of_type = building_type:grand_portal_ruins
			construct_building = {
				building_type = building_type:grand_portal
				cost_multiplier = 0
				cost_multiplier_reason = "trailer_staging"
				instant = yes
			}
		}

		research_advance = advance_type:grand_portal_restored

		set_global_variable = {
			name = portal_restored
			value = yes
		}

		remove_country_modifier = realm_lord
		add_country_modifier = {
			modifier = true_realm_lord
			mode = add_and_extend
			desc = you_are_truly_worthy
		}
	}
}

# Portal network everywhere: hub in capital, lesser node in every province capital
# (makes the portal_network map mode look alive for Shot 14)
elf_trailer.4 = {
	title = no_title
	desc = no_desc
	outcome = neutral
	type = country_event
	hidden = yes

	immediate = {
		capital = {
			construct_building = {
				building_type = building_type:prime_portal_hub
				cost_multiplier = 0
				cost_multiplier_reason = "trailer_staging"
				instant = yes
			}
		}

		every_province = {
			every_location_in_province = {
				limit = {
					is_province_capital = yes
					NOT = { has_building = building_type:grand_portal }
					NOT = { has_building = building_type:prime_portal_hub }
				}
				construct_building = {
					building_type = building_type:lesser_portal_node
					cost_multiplier = 0
					cost_multiplier_reason = "trailer_staging"
					instant = yes
				}
			}
		}
	}
}
